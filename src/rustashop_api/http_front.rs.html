1//! Serenade HTTP front controller (commerce routes migrate onto this over time).
2
3use serenade_http::{
4    box_future, AsyncHttpKernel, HttpError, Method, Request, Response, Route, RouteCollection,
5    UrlMatcher,
6};
7
8use crate::health::health_json_body;
9
10const HEALTHZ_ROUTE: &str = "healthz";
11
12/// Builds the Serenade async kernel used for HTTP that has already moved off Actix handlers.
13///
14/// Today: `GET /healthz`. Remaining commerce stays on Actix routes until later cutover PRs.
15#[must_use]
16pub fn commerce_http_kernel() -> AsyncHttpKernel {
17    let routes = healthz_matcher();
18    AsyncHttpKernel::from_async_fn(move |request: &mut Request| {
19        let outcome = routes.apply(request);
20        box_future(async move {
21            match outcome {
22                Ok(found) if found.route_name() == HEALTHZ_ROUTE => Ok(healthz_response()),
23                Ok(_) => Err(HttpError::not_found("no handler")),
24                Err(error) => Err(error),
25            }
26        })
27    })
28}
29
30fn healthz_matcher() -> UrlMatcher {
31    let mut collection = RouteCollection::new();
32    collection
33        .add(Route::with_method(HEALTHZ_ROUTE, "/healthz", Method::Get))
34        .expect("healthz route");
35    UrlMatcher::new(collection)
36}
37
38fn healthz_response() -> Response {
39    Response::new(200)
40        .with_header("content-type", "application/json")
41        .with_body(health_json_body())
42}
43
44/// Actix service that forwards to the Serenade kernel (used for `/healthz` during the shell).
45#[allow(clippy::future_not_send)]
46pub async fn serenade_dispatch(
47    request: actix_web::HttpRequest,
48    body: actix_web::web::Bytes,
49    kernel: actix_web::web::Data<AsyncHttpKernel>,
50) -> actix_web::HttpResponse {
51    serenade_http_actix::dispatch_async(kernel.get_ref(), &request, body).await
52}
53
54/// Registers Serenade-fronted routes on an Actix config (compose with leftover Actix commerce).
55pub fn configure_serenade_front(cfg: &mut actix_web::web::ServiceConfig) {
56    cfg.route("/healthz", actix_web::web::get().to(serenade_dispatch));
57}

59#[cfg(test)]
60mod tests {
61    use super::*;
62    use actix_web::test as actix_test;
63    use actix_web::{web, App};
64    use serenade_http::ROUTE_ATTRIBUTE;
65
66    use crate::health::HealthResponse;
67
68    #[actix_web::test]
69    async fn healthz_via_serenade_kernel() {
70        let kernel = web::Data::new(commerce_http_kernel());
71        let app = actix_test::init_service(
72            App::new()
73                .app_data(kernel)
74                .configure(configure_serenade_front),
75        )
76        .await;
77        let req = actix_test::TestRequest::get().uri("/healthz").to_request();
78        let resp = actix_test::call_service(&app, req).await;
79        assert!(resp.status().is_success());
80        let body: HealthResponse = actix_test::read_body_json(resp).await;
81        assert_eq!(body.status, "ok");
82        assert_eq!(body.kernel, rustashop::kernel_status());
83    }
84
85    #[actix_web::test]
86    async fn kernel_rejects_unknown_path() {
87        let kernel = commerce_http_kernel();
88        let response = kernel.handle(Request::new(Method::Get, "/nope")).await;
89        assert_eq!(response.status(), 404);
90    }
91
92    #[test]
93    fn matcher_sets_route_attribute() {
94        let matcher = healthz_matcher();
95        let mut request = Request::new(Method::Get, "/healthz");
96        let found = matcher.apply(&mut request).expect("match");
97        assert_eq!(found.route_name(), HEALTHZ_ROUTE);
98        assert_eq!(
99            request
100                .attributes()
101                .get::<String>(ROUTE_ATTRIBUTE)
102                .map(String::as_str),
103            Some(HEALTHZ_ROUTE)
104        );
105    }
