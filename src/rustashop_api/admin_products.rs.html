1//! Admin product list.
2
3use actix_web::{get, web, HttpResponse};
4use rustashop_persist::CatalogRepository;
5use serde::Deserialize;
6use serenade_contracts::PageRequest;
7use utoipa::IntoParams;
8
9use crate::admin_auth::{AdminAuthConfig, AdminBearer};
10use crate::error::{ApiError, ErrorBody};
11use crate::products::{ProductListResponse, ProductResponse};
12
13const DEFAULT_LIMIT: u32 = 20;
14const MAX_LIMIT: u32 = 100;
15
16/// Query string for admin product list.
17#[derive(Debug, Deserialize, IntoParams)]
18pub struct ListAdminProductsQuery {
19    /// Maximum rows (capped).
20    pub limit: Option<u32>,
21    /// Rows to skip.
22    pub offset: Option<u32>,
23}
24
25/// `GET /v1/{admin_api_prefix}/products` - list products including disabled (bearer required).
26#[utoipa::path(
27    get,
28    path = "/v1/{admin_api_prefix}/products",
29    params(ListAdminProductsQuery),
30    security(("admin_bearer" = [])),
31    responses(
32        (status = 200, description = "Product page", body = ProductListResponse),
33        (status = 401, description = "Missing or invalid bearer", body = ErrorBody)
34    )
35)]
36#[get("/products")]
37pub async fn list_admin_products(
38    bearer: AdminBearer,
39    auth: web::Data<AdminAuthConfig>,
40    catalog: web::Data<CatalogRepository>,
41    query: web::Query<ListAdminProductsQuery>,
42) -> Result<HttpResponse, ApiError> {
43    auth.authorize_bearer(bearer.0.as_deref())?;
44    let limit = query.limit.unwrap_or(DEFAULT_LIMIT).clamp(1, MAX_LIMIT);
45    let offset = query.offset.unwrap_or(0);
46    let page = PageRequest { limit, offset };
47    let products = catalog
48        .list_all_products(page)
49        .await
50        .map_err(|error| ApiError::from_persist(&error))?;
51    Ok(HttpResponse::Ok().json(ProductListResponse {
52        items: products.into_iter().map(ProductResponse::from).collect(),
53    }))
54}
