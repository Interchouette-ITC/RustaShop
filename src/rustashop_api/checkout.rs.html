42/// Order JSON returned by checkout.
43#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq, ToSchema)]
44pub struct OrderResponse {
45    /// Order id.
46    pub id: String,
47    /// Human-readable number.
48    pub number: String,
49    /// Source cart id.
50    pub cart_id: Option<String>,
51    /// Fulfillment state.
52    pub state: String,
53    /// Payment status (`pending` at checkout; no provider attached yet).
54    pub payment_status: String,
55    /// Order currency.
56    pub currency: String,
57    /// Sum of line totals.
58    pub items_total: MoneyResponse,
59    /// Payable total.
60    pub total: MoneyResponse,
61    /// Order lines.
62    pub lines: Vec<OrderLineResponse>,
63}

65impl From<Order> for OrderResponse {
66    fn from(order: Order) -> Self {
67        Self {
68            id: order.id,
69            number: order.number,
70            cart_id: order.cart_id,
71            state: order.state,
72            payment_status: order.payment_status,
73            currency: order.currency.as_str().to_owned(),
74            items_total: MoneyResponse {
75                amount_minor: order.items_total.amount_minor,
76                currency: order.items_total.currency.as_str().to_owned(),
77            },
78            total: MoneyResponse {
79                amount_minor: order.total.amount_minor,
80                currency: order.total.currency.as_str().to_owned(),
81            },
82            lines: order
83                .lines
84                .into_iter()
85                .map(|line| OrderLineResponse {
86                    id: line.id,
87                    variant_id: line.variant_id,
88                    quantity: line.quantity,
89                    unit_price: MoneyResponse {
90                        amount_minor: line.unit_price.amount_minor,
91                        currency: line.unit_price.currency.as_str().to_owned(),
92                    },
93                    line_total: MoneyResponse {
94                        amount_minor: line.line_total.amount_minor,
95                        currency: line.line_total.currency.as_str().to_owned(),
96                    },
97                    product_name: line.product_name,
98                    variant_sku: line.variant_sku,
99                })
100                .collect(),
101        }
102    }
103}

105struct IdempotencyKey(Option<String>);
106
107impl FromRequest for IdempotencyKey {
108    type Error = actix_web::Error;
109    type Future = Ready<Result<Self, Self::Error>>;
110
111    fn from_request(request: &HttpRequest, _payload: &mut Payload) -> Self::Future {
112        ready(Ok(Self(idempotency_key(request))))
113    }
114}
115
116fn idempotency_key(request: &HttpRequest) -> Option<String> {
117    request
118        .headers()
119        .get("idempotency-key")
120        .and_then(|value| value.to_str().ok())
121        .map(str::trim)
122        .filter(|value| !value.is_empty())
123        .map(ToOwned::to_owned)
124}

126/// Converts a cart into a placed order.
127#[utoipa::path(
128    post,
129    path = "/v1/checkout",
130    request_body = CheckoutRequest,
131    params(("Idempotency-Key" = Option<String>, Header, description = "Replay token")),
132    responses(
133        (status = 201, description = "Order placed", body = OrderResponse),
134        (status = 404, description = "Unknown cart", body = ErrorBody),
135        (status = 409, description = "Cart already checked out", body = ErrorBody),
136        (status = 422, description = "Empty cart", body = ErrorBody)
137    )
138)]
139#[post("/v1/checkout")]
140pub async fn place_order(
141    store: web::Data<CatalogRepository>,
142    key: IdempotencyKey,
143    body: web::Json<CheckoutRequest>,
144) -> Result<HttpResponse, ApiError> {
145    ensure_request_param(&body.cart_id)?;
146    let key = ensure_request_param_opt(key.0.as_deref())?;
147    let order = store
148        .checkout_cart(&body.cart_id, key)
149        .await
150        .map_err(|error| ApiError::from_persist(&error))?;
151    Ok(HttpResponse::Created().json(OrderResponse::from(order)))
152}
